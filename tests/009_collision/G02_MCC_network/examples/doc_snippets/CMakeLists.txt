cmake_minimum_required(VERSION 3.20)
project(G02_documented_callers LANGUAGES C CXX Fortran)
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/BuildG02.cmake")
add_executable(doc_batch batch.cpp)
target_link_libraries(doc_batch PRIVATE g02_mcc)
add_executable(doc_c minimal.c)
target_link_libraries(doc_c PRIVATE g02_mcc)
add_executable(doc_fortran minimal.f90)
target_link_libraries(doc_fortran PRIVATE g02_mcc_fortran)
enable_testing()
set(model "${CMAKE_CURRENT_LIST_DIR}/../../data/collision_box_elastic")
add_test(NAME documented_cpp COMMAND doc_batch "${model}")
add_test(NAME documented_c COMMAND doc_c "${model}")
add_test(NAME documented_fortran COMMAND doc_fortran "${model}")
